# matplotlib GUI が生成したスクリプト
# 使い方: データファイル「datetime.csv」と同じフォルダに置いて、python plot.py で実行します。
# 必要なライブラリ: pandas, matplotlib, numpy
import matplotlib.dates as mdates
import matplotlib.pyplot as plt
import numpy as np
import pandas as pd

# 日本語フォント: 一覧の先頭から、見つかったものを使う。
# どれも無い環境では DejaVu Sans になる（日本語は豆腐になるが、実行はできる）
plt.rcParams["font.family"] = "sans-serif"
plt.rcParams["font.sans-serif"] = ["Noto Sans JP", "Noto Sans CJK JP", "IPAexGothic", "Yu Gothic", "Hiragino Sans", "MS Gothic", "DejaVu Sans"]
plt.rcParams["axes.unicode_minus"] = False
FONT_SIZE = 15.0  # 文字の大きさ（pt）

# ==== 1. データの読み込み ====
# 形式: CSV（.csv）
DATA_FILE = "datetime.csv"
df = pd.read_csv(
    DATA_FILE,
    encoding="utf-8",  # 文字コード（自動判定）
    sep=",",  # 区切り文字
    header=0,  # 先頭行をヘッダにする（None: ヘッダなし）
    skiprows=0,  # ヘッダより前に読み飛ばす行数
    thousands=None,  # 桁区切り（None: なし）
    decimal=".",  # 小数点
    comment=None,  # コメント記号（None: なし）。この記号から行末までを読まない
    engine="python",
)
# 日時の列を日時に変換する（自動で認識した書式。読めない値は NaT（欠損）になる）
df.isetitem(0, pd.to_datetime(df.iloc[:, 0], format="%Y/%m/%d %H:%M", errors="coerce"))  # 日時 [0]

# ==== 2. 描画に使う行 ====
# 先頭の行は除外しない（全ての行を使う）

# ==== 3. 図と軸 ====
fig, ax = plt.subplots(figsize=(8.0, 6.0), dpi=100)  # 幅 8.0 × 高さ 6.0 インチ

# ==== 4. 系列 ====
# 系列1: X = 日時 [0]、Y = 温度 [3]
x = df.iloc[:, 0]  # 日時の列（読み込み時に日時に変換済み）
y = pd.to_numeric(df.iloc[:, 3], errors="coerce")  # 数値にできない値は NaN にする
ok = x.notna() & y.notna()  # X か Y が欠けている行は描かない
ax.plot(x[ok], y[ok], color="#FF4B00", linewidth=2.0, linestyle="solid", label="温度 [3]")

# ==== 5. 軸（ラベル・スケール・範囲・目盛） ====
ax.set_xlabel("日時", fontsize=FONT_SIZE)
ax.set_ylabel("温度 [3]", fontsize=FONT_SIZE)
# X が日時の列: 範囲に合わせて目盛の間隔と表示を自動で決める
date_locator = mdates.AutoDateLocator()
ax.xaxis.set_major_locator(date_locator)
ax.xaxis.set_major_formatter(mdates.ConciseDateFormatter(date_locator))
# 目盛は内向きにする
ax.tick_params(which="both", direction="in", labelsize=FONT_SIZE - 1)

# ==== 6. グリッドと凡例 ====
# グリッドは表示しない
ax.legend(loc="best", fontsize=FONT_SIZE - 1, edgecolor="black")

# ==== 7. 余白 ====
# 余白は自動で調整する
fig.tight_layout()

# ==== 8. 保存と表示 ====
fig.savefig("plot.png", dpi=300)  # GUI の「保存」と同じ形式（png）と解像度（300 dpi）
plt.show()
